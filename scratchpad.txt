    #include <SPI.h>
    #include <TFT_eSPI.h>

    // Create a TFT_eSPI instance:
    TFT_eSPI tft = TFT_eSPI(); 
    TFT_eSprite img = TFT_eSprite(&tft); // Create a Sprite object
    int angle = 45;

    void setup() {
      Serial.begin(115200);
      tft.init();
      tft.fillScreen(TFT_BLACK);
      tft.setPivot(tft.width()/2, tft.height()/2); // Change just the TFT pivot point

      img.createSprite(80, 80);
      img.fillSprite(TFT_BLACK); 
      img.setTextColor(TFT_WHITE, TFT_BLACK);
      img.setTextSize(3);
      
      img.setTextDatum(MC_DATUM);
      img.drawString("A", 40, 40);
      img.pushRotated(angle, TFT_BLACK); 
    }

    void loop() {
      angle = angle + 5;
      if (angle >= 360) angle = 0;
      img.pushRotated(angle, TFT_BLACK);
      delay(30);
    }
